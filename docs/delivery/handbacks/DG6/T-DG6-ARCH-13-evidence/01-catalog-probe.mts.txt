import { spawnSync } from "node:child_process";
import { createRequire } from "node:module";
const root = process.argv[2]!;
const gs = await import(root + "/packages/db/test/global-setup.ts");
const pg = createRequire(root + "/packages/db/package.json")("pg");
const admin = process.env.TEST_DATABASE_ADMIN_URL!;
const db = await gs.createScratchDatabase(admin, "arch13");
const owner = gs.roleUrl(admin, db, "mth_owner");
const r = spawnSync(process.execPath, ["--conditions=@mth/source", root + "/packages/db/src/cli.ts", "migrate"], { env: { ...process.env, NODE_ENV: "test", DATABASE_OWNER_URL: owner }, encoding: "utf8" });
console.log("migrate exit", r.status, (r.stdout.match(/applying/g) ?? []).length, "applied;", r.stderr.slice(-400));
const c = new pg.Client({ connectionString: gs.roleUrl(admin, db, null) }); await c.connect();
const q = async (s: string) => (await c.query(s)).rows;
console.log("LAST MIGRATIONS"); for (const x of await q(`SELECT * FROM schema_migration ORDER BY 1 DESC LIMIT 3`).catch(async () => [])) console.log(JSON.stringify(x));
console.log("COLUMNS"); for (const x of await q(`SELECT column_name, data_type, is_nullable, column_default, ordinal_position FROM information_schema.columns WHERE table_name='export_job' AND column_name IN ('attempt_count','assembly_attempt_count','snapshot_id','selection') ORDER BY ordinal_position`)) console.log(JSON.stringify(x));
console.log("CONSTRAINTS"); for (const x of await q(`SELECT conname, contype, pg_get_constraintdef(oid) d FROM pg_constraint WHERE conrelid='export_job'::regclass ORDER BY 1`)) console.log(x.conname, x.contype, "=>", x.d);
console.log("INDEXES"); for (const x of await q(`SELECT indexdef FROM pg_indexes WHERE tablename='export_job' ORDER BY indexname`)) console.log(x.indexdef);
console.log("TRIGGERS"); for (const x of await q(`SELECT tgname, pg_get_triggerdef(oid) d FROM pg_trigger WHERE tgrelid='export_job'::regclass AND NOT tgisinternal ORDER BY 1`)) console.log(x.tgname, "=>", x.d);
console.log("GRANTS"); for (const x of await q(`SELECT privilege_type FROM information_schema.role_table_grants WHERE table_name='export_job' AND grantee='mth_app' ORDER BY 1`)) console.log(x.privilege_type);
console.log("GUARD CONSTRAINT NAMES (from prosrc)");
const src = (await q(`SELECT prosrc, proconfig, prosecdef FROM pg_proc WHERE proname='export_job_guard'`))[0];
console.log("config", JSON.stringify(src.proconfig), "secdef", src.prosecdef);
console.log([...new Set([...src.prosrc.matchAll(/CONSTRAINT = '([a-z_]+)'/g)].map((m: any) => m[1]))].join(" "));
console.log("insert rule:", src.prosrc.match(/IF NEW.status <> 'queued'[^\n]*/)?.[0]);
console.log("definition check:", src.prosrc.match(/IF NEW.selection IS NOT NULL AND s.definition_code[^\n]*/)?.[0]);
await c.end(); await gs.dropScratchDatabase(admin, db);
