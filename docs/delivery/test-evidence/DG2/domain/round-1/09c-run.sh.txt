#!/usr/bin/env bash
# Domain-reviewer live scenario (DG2 round 1). Disposable PG16 cluster + disposable clone; SYNTHETIC data only.
set -uo pipefail
SRC="$TMPDIR/review-dg2"; cd "$SRC"
source "$TMPDIR/scn/pg.sh" 54395 "$TMPDIR/scn/work"; WORK="$TMPDIR/scn/work"
cleanup() { for p in ${API_PID:-} ${WPID:-}; do kill $p 2>/dev/null; done; [ -n "${PG_PID:-}" ] && kill -INT $PG_PID 2>/dev/null; sleep 2; }
trap cleanup EXIT
DB="postgresql://postgres@127.0.0.1:${PG_PORT}/mth"
echo "== env: $(date -u +%FT%TZ) node $(node -v) $($PGBIN/postgres --version) commit $(git rev-parse HEAD)"
psql "$ADMIN" -q -v ON_ERROR_STOP=1 -c "CREATE ROLE mth_owner NOLOGIN" -c "CREATE ROLE mth_app NOLOGIN" -c "CREATE DATABASE mth OWNER mth_owner"
role_url() { node -e 'const u=new URL(process.argv[1]);u.pathname="/mth";u.searchParams.set("options","-c role="+process.argv[2]);console.log(u.toString())' "$ADMIN" "$1"; }
export NODE_ENV=development AUTH_MODE=dev PORT=3000 APP_BASE_URL=http://localhost:3000 LOG_LEVEL=warn
export DATABASE_OWNER_URL="$(role_url mth_owner)" DATABASE_URL="$(role_url mth_app)" AUTH_RATE_LIMIT_PER_MINUTE=1000 RATE_LIMIT_PER_MINUTE=100000
echo "== migrate"; node packages/db/dist/cli.js migrate; echo "exit=$?"
echo "== seed-dev (SYNTHETIC)"; node packages/db/dist/cli.js seed-dev; echo "exit=$?"
node apps/api/dist/main.js >"$WORK/api.log" 2>&1 & API_PID=$!
for _ in $(seq 1 60); do curl -fsS http://localhost:3000/readyz >/dev/null 2>&1 && break; sleep 0.5; done
echo "== /readyz: $(curl -s -w ' %{http_code}' http://localhost:3000/readyz)"
echo "== API probe"; PLAYBOOK_BLOCKS="$SRC/docs/source/playbook.blocks.json" PLAYBOOK_MD="$SRC/docs/source/playbook.md" node "$TMPDIR/scn/probe.mjs" | tee "$WORK/probe.out"; echo "probe exit=${PIPESTATUS[0]}"
echo "== probe2"; node "$TMPDIR/scn/probe2.mjs" | tee "$WORK/probe2.out"; echo "probe2 exit=${PIPESTATUS[0]}"
TID=$(grep -o '"transformationId":"[^"]*"' "$WORK/probe.out" | tail -n1 | cut -d'"' -f4)
echo "== audit events for the scenario transformation (gate + charter + decisions):"
psql "$DB" -qAtc "select action, count(*) from audit_event where action like 'gate%' or action like 'charter%' or action like 'decision%' or action like 'north_star%' or action like 'tom_workshop%' group by 1 order by 1" | tr '\n' ' '; echo
echo "== engineering gate tables touched? (expect none named dg/delivery):"; psql "$DB" -qAtc "select count(*) from information_schema.tables where table_schema='public' and (table_name ilike '%delivery%' or table_name ~* '^dg')"
if [ "${SCREENS:-0}" = "1" ]; then echo "== screens"; cp "$TMPDIR/scn/screens.mjs" ./zz-screens.mjs && PLAYWRIGHT_BROWSERS_PATH=/opt/pw-browsers TID="$TID" node zz-screens.mjs "$1"; echo "screens exit=$?"; rm -f zz-screens.mjs; fi
echo "== api.log error/fatal lines: $(grep -c '"level":[56]0' "$WORK/api.log")"
