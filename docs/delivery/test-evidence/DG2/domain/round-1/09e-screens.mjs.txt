// Domain-reviewer screens (DG2 round 1). Synthetic data only. Headless Chromium (pre-installed, /opt/pw-browsers).
import { chromium } from "@playwright/test";
const OUT = process.argv[2];
const BASE = "http://localhost:3000";
const TID = process.env.TID;
const browser = await chromium.launch();
const res = [];
const settle = async (page) => { await page.waitForLoadState("networkidle"); await page.waitForTimeout(900); };
const pages = [
  ["create", "/transformations/new"],
  ["workspace", `/transformations/${TID}`],
  ["diagnose", `/transformations/${TID}/diagnose`],
  ["charter", `/transformations/${TID}/charter`],
  ["define", `/transformations/${TID}/define`],
  ["design", `/transformations/${TID}/design`],
  ["gate-g1", `/transformations/${TID}/gates/G1`],
  ["gate-g2", `/transformations/${TID}/gates/G2`],
  ["team", `/transformations/${TID}/team`],
];
for (const pref of ["en", "ar"]) {
  const ctx = await browser.newContext({ locale: pref === "ar" ? "ar-SA" : "en-US", timezoneId: "Asia/Riyadh", viewport: { width: 1366, height: 900 } });
  const page = await ctx.newPage(); const foreign = []; const consoleErrors = [];
  page.on("request", (r) => { const u = new URL(r.url()); if (u.hostname !== "localhost") foreign.push(r.url()); });
  page.on("console", (m) => { if (m.type() === "error") consoleErrors.push(m.text().slice(0, 200)); });
  await page.goto(BASE + "/login"); await settle(page);
  await page.evaluate(async (p) => {
    await fetch("/api/v1/auth/dev-login", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ username: "dev.lead" }) });
    const me = await fetch("/api/v1/me"); const body = await me.json();
    await fetch("/api/v1/me/preferences", { method: "PUT", headers: { "content-type": "application/json", "if-match": me.headers.get("etag"), "x-csrf-token": body.csrfToken }, body: JSON.stringify({ preferredLocale: p }) });
  }, pref);
  for (const [name, path] of pages) {
    await page.goto(BASE + path); await settle(page);
    if (name === "create") { const r = page.locator("input[type=radio]").nth(1); if (await r.count()) { await r.check().catch(() => {}); await page.waitForTimeout(400); } }
    const info = await page.evaluate(() => ({
      lang: document.documentElement.lang, dir: document.documentElement.dir,
      text: document.body.innerText.replace(/\s+/g, " ").slice(0, 2500),
      rawKeys: (document.body.innerText.match(/\b[a-z]+(\.[a-zA-Z_]+){2,}\b/g) ?? []).slice(0, 10),
    }));
    await page.screenshot({ path: `${OUT}/screen-${name}-${pref}.png`, fullPage: true });
    res.push({ pref, name, ...info });
  }
  res.push({ pref, foreignRequests: foreign, consoleErrors });
  await ctx.close();
}
await browser.close();
console.log(JSON.stringify(res, null, 1));
