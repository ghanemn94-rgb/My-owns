import { chromium } from "@playwright/test";
const OUT = process.argv[2];
const BASE = "http://localhost:3000";
const browser = await chromium.launch();
let fail = 0;
const res = [];
for (const [user, locale] of [["dev.office", "ar-SA"], ["dev.auditor", "en-US"]]) {
  const ctx = await browser.newContext({ locale, timezoneId: "Asia/Riyadh", viewport: { width: 1366, height: 860 } });
  const page = await ctx.newPage();
  const foreign = [];
  page.on("request", (r) => { const u = new URL(r.url()); if (u.hostname !== "localhost") foreign.push(r.url()); });
  await page.goto(BASE + "/login");
  await page.waitForLoadState("networkidle");
  const tag = locale.slice(0, 2);
  const loginAttrs = await page.evaluate(() => ({ lang: document.documentElement.lang, dir: document.documentElement.dir, text: document.body.innerText.slice(0, 600) }));
  await page.screenshot({ path: `${OUT}/screen-login-${tag}.png`, fullPage: true });
  // sign in through the UI dev form via API in page context (same origin)
  await page.evaluate(async (u) => { await fetch("/api/v1/auth/dev-login", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ username: u }) }); }, user);
  await page.goto(BASE + "/");
  await page.waitForLoadState("networkidle");
  await page.waitForTimeout(800);
  const shell = await page.evaluate(() => ({ lang: document.documentElement.lang, dir: document.documentElement.dir, text: document.body.innerText.slice(0, 1500), imgs: document.querySelectorAll("header img, header svg").length, font: getComputedStyle(document.body).fontFamily }));
  await page.screenshot({ path: `${OUT}/screen-home-${tag}.png`, fullPage: true });
  await page.goto(BASE + "/transformations");
  await page.waitForLoadState("networkidle");
  await page.waitForTimeout(800);
  const list = await page.evaluate(() => ({ dir: document.documentElement.dir, text: document.body.innerText.slice(0, 2000) }));
  await page.screenshot({ path: `${OUT}/screen-transformations-${tag}.png`, fullPage: true });
  const fontsLoaded = await page.evaluate(async () => { await document.fonts.ready; return [...document.fonts].filter((f) => f.status === "loaded").map((f) => f.family); });
  res.push({ user, locale, loginAttrs, shell, list, fontsLoaded: [...new Set(fontsLoaded)], foreignRequests: foreign });
  await ctx.close();
}
await browser.close();
console.log(JSON.stringify(res, null, 2));
