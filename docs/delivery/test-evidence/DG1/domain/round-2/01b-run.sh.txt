#!/usr/bin/env bash
# Domain-reviewer live scenario (DG1 round 2). Disposable PG16 cluster + disposable clone; SYNTHETIC data only.
set -uo pipefail
SRC="$TMPDIR/review-dg1r2"; OUT="$1"; cd "$SRC"
source "$TMPDIR/scn/pg.sh" 54393 "$TMPDIR/scn/work"; WORK="$TMPDIR/scn/work"
cleanup() { for p in ${API_PID:-} ${API2_PID:-} ${WPID:-}; do kill $p 2>/dev/null; done; [ -n "${PG_PID:-}" ] && kill -INT $PG_PID 2>/dev/null; sleep 2; }
trap cleanup EXIT
DB="postgresql://postgres@127.0.0.1:${PG_PORT}/mth"
echo "== env: $(date -u +%FT%TZ) node $(node -v) $($PGBIN/postgres --version) commit $(git rev-parse HEAD)"
psql "$ADMIN" -q -v ON_ERROR_STOP=1 -c "CREATE ROLE mth_owner NOLOGIN" -c "CREATE ROLE mth_app NOLOGIN" -c "CREATE DATABASE mth OWNER mth_owner"
role_url() { node -e 'const u=new URL(process.argv[1]);u.pathname="/mth";u.searchParams.set("options","-c role="+process.argv[2]);console.log(u.toString())' "$ADMIN" "$1"; }
export NODE_ENV=development AUTH_MODE=dev PORT=3000 APP_BASE_URL=http://localhost:3000 LOG_LEVEL=warn
export DATABASE_OWNER_URL="$(role_url mth_owner)" DATABASE_URL="$(role_url mth_app)" AUTH_RATE_LIMIT_PER_MINUTE=1000 RATE_LIMIT_PER_MINUTE=10000
echo "== migrate (fresh db)"; node packages/db/dist/cli.js migrate; echo "exit=$?"
echo "== migrate again (no-op)"; node packages/db/dist/cli.js migrate; echo "exit=$?"
psql "$DB" -qAtc "select count(*)||' migrations recorded: '||string_agg(version::text, ',' order by version) from schema_migration" 2>&1 || psql "$DB" -qAtc "select count(*)||' migrations recorded' from schema_migration"
echo "== public views:"; psql "$DB" -qAtc "select table_name from information_schema.views where table_schema='public' order by 1" | tr '\n' ' '; echo
echo "== view columns (compare with data-dictionary.md 'Views'):"; psql "$DB" -qAtc "select table_name||'.'||column_name||':'||data_type from information_schema.columns where table_name in ('actor_display','business_unit_closure','scope_node') order by table_name, ordinal_position" | tr '\n' ' '; echo
echo "== mth_app privileges on views:"; psql "$DB" -qAtc "select table_name||':'||string_agg(privilege_type, ',') from information_schema.role_table_grants where grantee='mth_app' and table_name in ('actor_display','business_unit_closure','scope_node') group by table_name order by 1" | tr '\n' ' '; echo
echo "== trigger on business_unit:"; psql "$DB" -qAtc "select tgname from pg_trigger where tgrelid='business_unit'::regclass and not tgisinternal" | tr '\n' ' '; echo
echo "== seed-dev (SYNTHETIC)"; node packages/db/dist/cli.js seed-dev; echo "exit=$?"
node apps/api/dist/main.js >"$WORK/api.log" 2>&1 & API_PID=$!
node apps/worker/dist/main.js >"$WORK/worker.log" 2>&1 & WPID=$!
for _ in $(seq 1 60); do curl -fsS http://localhost:3000/readyz >/dev/null 2>&1 && break; sleep 0.5; done
echo "== /healthz: $(curl -s http://localhost:3000/healthz)"; echo "== /readyz: $(curl -s -w ' %{http_code}' http://localhost:3000/readyz)"
echo "== index.html head:"; curl -s http://localhost:3000/ | head -c 600; echo
echo; echo "== API probe"; node "$TMPDIR/scn/probe.mjs"; echo "probe exit=$?"
echo "== DB-level cycle guard (owner role bypasses the API): direct UPDATE making SYN-OPS a child of its descendant SYN-RETAIL"
psql "$DB" -v ON_ERROR_STOP=1 -qAtc "SET ROLE mth_app; UPDATE business_unit SET parent_business_unit_id='01920000-0000-7000-9000-000000000102' WHERE id='01920000-0000-7000-9000-000000000101'" 2>&1; echo "psql exit=$?"
echo "== audit events for BU moves and denials:"; psql "$DB" -qAtc "select action, count(*) from audit_event where action in ('business_unit.update','business_unit.create','authorization.denied','scoped_assignment.create') group by 1 order by 1" | tr '\n' ' '; echo
psql "$DB" -qAtc "select action, record_type, left(coalesce(reason,'')||' '||coalesce(changes::text, ''),200) from audit_event where action='authorization.denied' order by seq desc limit 3" 2>&1
echo "== closure view has no self-cycles: $(psql "$DB" -qAtc "select count(*) from business_unit_closure where ancestor_id=descendant_id and depth>0")"
sleep 3
echo "== outbox after worker run:"; psql "$DB" -qAtc "select event_type, count(*), count(published_at) from outbox_event group by 1" 2>&1
echo; echo "== worker stop test: SIGTERM worker pid $WPID"
kill -TERM $WPID; wait $WPID; echo "worker exit=$?"; WPID=
sleep 1; echo "API alive: $(kill -0 $API_PID && echo yes)"; echo "/readyz after worker stopped: $(curl -s -w ' %{http_code}' http://localhost:3000/readyz)"
echo "worker log tail:"; tail -3 "$WORK/worker.log"
echo; echo "== rate-limit keying: second API instance on :3001 with RATE_LIMIT_PER_MINUTE=20"
PORT=3001 APP_BASE_URL=http://localhost:3001 RATE_LIMIT_PER_MINUTE=20 node apps/api/dist/main.js >"$WORK/api2.log" 2>&1 & API2_PID=$!
for _ in $(seq 1 60); do curl -fsS http://localhost:3001/readyz >/dev/null 2>&1 && break; sleep 0.5; done
node "$TMPDIR/scn/ratelimit.mjs"; echo "ratelimit exit=$?"
kill $API2_PID; API2_PID=
echo; echo "== AR/EN screens"; cp "$TMPDIR/scn/screens.mjs" ./zz-screens.mjs && PLAYWRIGHT_BROWSERS_PATH=/opt/pw-browsers node zz-screens.mjs "$OUT"; echo "screens exit=$?"; rm -f zz-screens.mjs
echo "== api.log error/fatal lines: $(grep -c '"level":[56]0' "$WORK/api.log")"
# usage: source pg.sh <port> <workdir>
PGBIN=/usr/lib/postgresql/16/bin; PG_PORT=$1; PGW=$2; rm -rf "$PGW"; mkdir -p "$PGW"; chmod 777 "$PGW"
as_pg() { if [ "$(id -u)" = "0" ]; then unshare --user --map-user=1000 --map-group=1000 "$@"; else "$@"; fi; }
as_pg $PGBIN/initdb -D "$PGW/pg" -U postgres --auth=trust >/dev/null
as_pg $PGBIN/postgres -D "$PGW/pg" -c unix_socket_directories='' -c listen_addresses=127.0.0.1 -p $PG_PORT -c fsync=off >"$PGW/pg.log" 2>&1 &
PG_PID=$!
ADMIN="postgresql://postgres@127.0.0.1:${PG_PORT}/postgres"
for _ in $(seq 1 60); do psql "$ADMIN" -qAtc "select 1" >/dev/null 2>&1 && break; sleep 0.5; done
