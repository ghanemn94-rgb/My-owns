const F = join(MODULES_DIR, "transformations", "planted.ts");

const log = (name: string, out: string[]) => console.log(`[plant] ${name} -> ${out.length} violation(s)${out.length ? ": " + out.join(" | ") : ""}`);

describe("F-DG1-127 A1 closed (code-security r7)", () => {
  it("bareAllowed", () => {
    for (const s of ["sqlite", "node:sqlite", "node:process", "process"]) { console.log(`[plant] bareAllowed(${s}) = ${bareAllowed(s)}`); expect(bareAllowed(s)).toBe(false); }
    for (const s of ["node:fs", "node:path", "node:crypto"]) { console.log(`[plant] bareAllowed(${s}) = ${bareAllowed(s)}`); expect(bareAllowed(s)).toBe(true); }
  });
  const must: [string, string][] = [
    ["A1 exact (assignment)", `import { DatabaseSync } from "node:sqlite";\nnew DatabaseSync(":memory:", { allowExtension: true }).loadExtension("/tmp/x.so");`],
    ["A1 bare sqlite", `import { DatabaseSync } from "sqlite";\nnew DatabaseSync(":memory:", { allowExtension: true }).loadExtension("/tmp/x.so");`],
    ["A1 namespace", `import * as s from "node:sqlite";\nnew s.DatabaseSync(":memory:", { allowExtension: true }).loadExtension("/tmp/x.so");`],
