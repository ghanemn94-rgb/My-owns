#!/bin/bash
# usage: withpg.sh <cmd...>  — starts the disposable PG on 54391, runs cmd, stops PG
PGBIN=/usr/lib/postgresql/16/bin
unshare --user --map-user=1000 --map-group=1000 $PGBIN/postgres -D $TMPDIR/pgw/pg -c unix_socket_directories='' -c listen_addresses=127.0.0.1 -p 54391 -c fsync=off > $TMPDIR/pgw/pg.log 2>&1 & PGP=$!
for i in $(seq 1 40); do psql postgresql://postgres@127.0.0.1:54391/postgres -qAtc "select 1" >/dev/null 2>&1 && break; sleep 0.5; done
export TEST_DATABASE_ADMIN_URL=postgresql://postgres@127.0.0.1:54391/postgres
"$@"; rc=$?
kill -INT $PGP; wait
exit $rc
