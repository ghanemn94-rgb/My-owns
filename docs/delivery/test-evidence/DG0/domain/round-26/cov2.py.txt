import csv,json,re,collections
reg={r['req_id']:r for r in csv.DictReader(open('docs/delivery/requirements.csv',encoding='utf-8'))}
sc=list(csv.DictReader(open('docs/analysis/source-coverage.csv',encoding='utf-8')))
print(collections.Counter(r['disposition'] for r in sc))
pb=open('docs/source/playbook.md',encoding='utf-8').read()
for r in sc:
    if r['disposition']!='REQUIREMENT':
        m=re.search(r'^(.*?)<!-- %s -->'%r['block_id'],pb,re.M)
        print(r['block_id'],r['disposition'],'|',(m.group(1).strip()[:70] if m else '(table)'),'|',r['rationale'][:70])
# bidirectional
fwd=collections.defaultdict(set)
for r in sc:
    for q in filter(None,r['req_ids'].split(';')): fwd[q].add(r['block_id'])
cls=collections.Counter(r['class'] for r in reg.values()); print('classes',cls)
bad=[]
for k,r in reg.items():
    refs=set(re.findall(r'B\d{4}',r['source_ref']))
    if k.startswith('REQ-PB'):
        if r['class']!='SOURCE' or not refs: bad.append((k,'pb-class/ref'))
        if refs-fwd[k] and fwd[k]-refs: bad.append((k,'mismatch',refs^fwd[k]))
    elif r['class']=='SOURCE': bad.append((k,'non-PB SOURCE'))
    if any(int(b[1:])>165 or int(b[1:])<1 for b in refs): bad.append((k,'bad block'))
pbrows=[k for k in reg if k.startswith('REQ-PB')]
print('REQ-PB rows',len(pbrows),'unmapped PB',[k for k in pbrows if not fwd[k]]); print('bad',bad)
