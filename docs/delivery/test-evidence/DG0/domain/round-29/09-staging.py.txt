import csv,re,collections
R=list(csv.DictReader(open('docs/delivery/requirements.csv',encoding='utf-8')))
G=collections.Counter(r['final_gate'] for r in R); print('final_gate',sorted(G.items()))
bad=[]
for r in R:
    inc=[x for x in re.split(r'[;\s]+',r['increments']) if x]
    fg=r['final_gate']
    if not re.fullmatch(r'DG[0-7]',fg): bad.append((r['req_id'],'final_gate',fg)); continue
    if not inc or any(not re.fullmatch(r'P[0-7]',i) for i in inc): bad.append((r['req_id'],'increments',r['increments'])); continue
    if max(int(i[1]) for i in inc)!=int(fg[2]): bad.append((r['req_id'],'max increment != final gate',r['increments'],fg))
    if not r['acceptance'].strip(): bad.append((r['req_id'],'no acceptance'))
print('staging problems',len(bad),bad[:15])
# §21 alignment spot-checks: template -> stage
exp={'T01':'DG2','T02':'DG2','T03':'DG2','T04':'DG2','T05':'DG3','T06':'DG3','T07':'DG3','T08':'DG3','T09':'DG3','T10':'DG4','T11':'DG4','T12':'DG4','T13':'DG4','T14':'DG4','T15':'DG4','T16':'DG4'}
for t,g in exp.items():
    rows=[r for r in R if r['template_id']==t and 'Implement Template' in r['title']]
    for r in rows: print(t,r['req_id'],r['final_gate'],'OK' if r['final_gate']==g else 'MISMATCH expected '+g)
for k,g in {'REQ-PB-016':'DG2','REQ-PB-017':'DG2','REQ-PB-018':'DG2','REQ-PB-019':'DG3','REQ-PB-020':'DG4','REQ-PB-021':'DG4','REQ-PB-086':'DG5','REQ-PB-091':'DG5','REQ-PB-088':'DG5','REQ-PB-029':'DG2','REQ-PB-041':'DG2','REQ-PB-053':'DG3'}.items():
    r=next(x for x in R if x['req_id']==k); print(k,r['title'][:60],r['final_gate'],'OK' if r['final_gate']==g else 'MISMATCH expected '+g)
dg0=[r['req_id'] for r in R if r['final_gate']=='DG0']; print('DG0-final rows',len(dg0),dg0)
print('non-DLV DG0 rows',[x for x in dg0 if not x.startswith('REQ-DLV')])
# vague acceptance
vague=[r['req_id'] for r in R if re.fullmatch(r'\s*(works|done|implemented|as specified)\.?\s*',r['acceptance'],re.I) or len(r['acceptance'])<25]
print('short/vague acceptance',vague)
