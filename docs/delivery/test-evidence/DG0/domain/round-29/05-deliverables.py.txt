import json,re,html
B={b['id']:b for b in json.load(open('docs/source/playbook.blocks.json'))['blocks']} if isinstance(json.load(open('docs/source/playbook.blocks.json')),dict) else {b['id']:b for b in json.load(open('docs/source/playbook.blocks.json'))}
fi=open('docs/analysis/field-inventory.md',encoding='utf-8').read()
def n(s): return re.sub(r'\s+',' ',html.unescape(s).replace('<br>',' ').replace('’',"'")).strip()
FI=n(fi)
def cells(bid):
    b=B[bid]
    if b['type']=='table': return [n(c) for r in b['rows'] for c in r if n(c)]
    return [n(b['text'])]

    tot=0;miss=[]
