# code-security-reviewer, DG0 round 29: for each of the 19 DG0-final requirements, check register status / final_gate,
# that every evidence path exists, and that every quoted test title in the row appears as passing in MY OWN logs
# (node-tests.log: 'ok N - <title>'; python-unittest.log: '<name> (...) ... ok'). Run: python3 requirements-check.py.txt
import csv, re, os, sys
E = "docs/delivery/test-evidence/DG0/code-security/round-29"
IDS = [f"REQ-DLV-{n:03d}" for n in (1,2,3,4,6,7,13,15,16,17,19,20,22,23,26,29,32)] + ["REQ-S20-024", "REQ-S20-025"]
node = open(f"{E}/node-tests.log", encoding="utf-8").read()
py = open(f"{E}/python-unittest.log", encoding="utf-8").read()
ok_node = set(re.findall(r"^\s*ok \d+ - (.+?)\s*$", node, re.M)); bad_node = set(re.findall(r"^\s*not ok \d+ - (.+?)\s*$", node, re.M))
ok_py = set(re.findall(r"^(test_\w+) \(.*\) \.\.\. ok$", py, re.M)) | set(re.findall(r"^test_\w+ \((test_\w+)\.", py, re.M))  # method names + module names with ok tests
rows = {r["req_id"]: r for r in csv.DictReader(open("docs/delivery/requirements.csv", encoding="utf-8"))}
dg0_final = sorted(k for k, r in rows.items() if r.get("final_gate") == "DG0")
print(f"# register rows with final_gate=DG0: {len(dg0_final)}; assignment list == register set: {sorted(IDS) == dg0_final}")
print(f"# node log: {len(ok_node)} ok, {len(bad_node)} not ok; python log: {len(ok_py)} ok")
problems = 0
for i in IDS:
    r = rows[i]
    ev = [p.strip() for p in re.split(r"[;\s]+", r.get("evidence", "")) if p.strip()]
    files = [p.split("#")[0] for p in ev if "/" in p]
    missing = [p for p in files if not os.path.isfile(p)]
    text = " ".join(r.values())
    titles = re.findall(r"\"([^\"]{12,})\"", text) + re.findall(r"'([^']{12,}?)'(?=[\s,;.)]|$)", text) + re.findall(r"\b(test_\w+)\b", text)
    passed = [t for t in titles if t in ok_node or t in ok_py or any(o.startswith(t) for o in ok_node)]
    failed = [t for t in titles if t in bad_node]
    unmatched = [t for t in titles if t not in passed and t not in failed]
    st_ok = r["status"] == "IMPLEMENTED" and r["final_gate"] == "DG0"
    problems += (not st_ok) + len(missing) + len(failed)
    print(f"{i}: status={r['status']} final_gate={r['final_gate']} evidence_files={len(files)} missing={missing} quoted_titles={len(titles)} passed={len(passed)} failed={len(failed)} unmatched={len(unmatched)}")
    for t in unmatched: print(f"    unmatched (prose or other-suite citation, reviewed manually): {t[:110]}")
print(f"PROBLEMS={problems}")
sys.exit(1 if problems else 0)
