// Real-repo census: for every CLOSED_VERIFIED finding, is fix_revision an ancestor of (a) the verifying run's head,
// (b) the verifying round's source_commit (when present), (c) the gate-candidate commit 6db3b83? And for every
// review run of every round: does the head_commit_at_start content hash to the round candidate (metadata-only drift)?
import { execFileSync } from "node:child_process";
import { readFileSync, existsSync } from "node:fs";
import { candidateId, manifestFromRef } from "./tools/gates/lib/candidate.mjs";
const R = process.cwd();
const git = (...a) => { try { execFileSync("git", ["-C", R, ...a], { stdio: "ignore" }); return true; } catch { return false; } };
const J = (p) => JSON.parse(readFileSync(p, "utf8"));
const stage = J("docs/delivery/stages.json").stages.find((s) => s.id === "DG0");
const F = J("docs/delivery/findings.json").findings.filter((f) => f.status === "CLOSED_VERIFIED");
const GATE = "6db3b8388699b767032b4f92547ceb79edf74519";
let bad = 0, n = 0, dropped = 0;
for (const f of F) {
  n++;
  const run = f.verification.invocation_reference.run_id;
  const meta = J(`docs/delivery/runs/DG0/${run}/meta.json`);
  const vh = meta.head_commit_at_start;
  const rd = run.match(/-R(\d+)-/); const rn = rd ? Number(rd[1]) : null;
  const round = stage.review_rounds.find((r) => r.round === rn);
  const a = git("merge-base", "--is-ancestor", f.fix_revision, vh);
  const srcPresent = round && git("cat-file", "-e", `${round.source_commit}^{commit}`);
  const b = srcPresent ? git("merge-base", "--is-ancestor", f.fix_revision, round.source_commit) : null;
  const c = git("merge-base", "--is-ancestor", f.fix_revision, GATE);
  if (!a || !c) { bad++; console.log(`BAD ${f.id} run=${run} inHead=${a} inGate=${c}`); }
  if (a && b === false) { dropped++; console.log(`HEAD-ONLY ${f.id}: fix in run head but NOT in round ${rn} source_commit`); }
}
console.log(`closures checked: ${n}; not-in-run-head-or-gate: ${bad}; in-head-but-not-round-candidate: ${dropped}`);
let drift = 0, runs = 0, skipped = 0;
for (const r of stage.review_rounds) {
  const mp = `docs/delivery/candidates/DG0/${r.candidate_id.slice(7, 23)}.manifest.json`;
  const m = existsSync(mp) ? J(mp) : null;
  for (const rel of Object.values(r.records)) {
    if (!existsSync(rel)) continue;
    const run = J(rel).invocation_reference.run_id; runs++;
    const h = J(`docs/delivery/runs/DG0/${run}/meta.json`).head_commit_at_start;
    try {
      const cid = candidateId(manifestFromRef(R, h, m.spec), m.hash_algorithm || "mth-candidate-v1");
      if (cid !== r.candidate_id) { drift++; console.log(`DRIFT round ${r.round} ${run}: head ${h.slice(0,10)} content ${cid.slice(0,19)} != round ${r.candidate_id.slice(0,19)}`); }
    } catch (e) { skipped++; console.log(`SKIP round ${r.round} ${run}: ${String(e.message).split("\n")[0]}`); }
  }
}
console.log(`review runs: ${runs}; head content != round candidate: ${drift}; not computable: ${skipped}`);
