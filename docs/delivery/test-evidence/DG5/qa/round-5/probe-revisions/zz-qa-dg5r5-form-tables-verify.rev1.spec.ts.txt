interface Measure {
  W: number;
  pageOverflow: number;
  tables: { sel: string; rows: number; region: null | { tabIndex: number; role: string | null; name: string; left: number; right: number; scrolls: boolean } }[];
  escapes: string[];
  dialog: null | { left: number; right: number };
}

/** Measures `scope` (a CSS selector: the main element or a dialog). */
async function measure(page: Page, scope: string): Promise<Measure> {
  return page.evaluate((scope) => {
    const W = window.innerWidth;
    const root = document.querySelector<HTMLElement>(scope);
    const inScroller = (el: Element) => {
      for (let x = el.parentElement; x && x !== document.body; x = x.parentElement) {
        const ox = getComputedStyle(x).overflowX;
        if (ox === "auto" || ox === "scroll" || ox === "hidden" || ox === "clip") return true;
      }
      return false;
    };
    const escapes: string[] = [];
    for (const el of Array.from(root?.querySelectorAll<HTMLElement>("*") ?? [])) {
      const r = el.getBoundingClientRect();
      if (r.width === 0 || r.height === 0) continue;
      const cs = getComputedStyle(el);
      if (cs.visibility === "hidden" || cs.position === "absolute" && r.width <= 1) continue;
      if (inScroller(el)) continue;
      if (r.left < -1 || r.right > W + 1)
        escapes.push(`${el.tagName.toLowerCase()}.${(el.getAttribute("class") ?? "").trim().split(/\s+/).join(".")} [${Math.round(r.left)}..${Math.round(r.right)}] "${(el.textContent ?? "").trim().slice(0, 40)}"`);
    }
    const tables = Array.from(root?.querySelectorAll<HTMLTableElement>("table") ?? []).map((t, i) => {
      const wrap = t.closest<HTMLElement>(".table-wrap");
      const r = wrap?.getBoundingClientRect();
      return {
        sel: `${i}:${Array.from(t.attributes).filter((a) => a.name.startsWith("data-")).map((a) => a.name).join(",") || t.className}`,
        rows: t.tBodies[0]?.rows.length ?? 0,
        region: wrap && r
          ? { tabIndex: wrap.tabIndex, role: wrap.getAttribute("role"), name: (wrap.getAttribute("aria-label") ?? "").trim(), left: Math.round(r.left), right: Math.round(r.right), scrolls: wrap.scrollWidth > wrap.clientWidth + 1 }
          : null,
      };
    });
    const d = document.querySelector("[role='dialog']")?.getBoundingClientRect();
    return { W, pageOverflow: document.documentElement.scrollWidth - W, tables, escapes: escapes.slice(0, 12), dialog: d ? { left: Math.round(d.left), right: Math.round(d.right) } : null };
  }, scope);
}

/** The region of `table` scrolls with the arrow key towards the reading end. Returns |scrollLeft| after one key. */
async function keyboardScroll(page: Page, lang: Lang, table: Locator): Promise<number> {
  const wrap = table.locator("xpath=ancestor::div[contains(concat(' ', normalize-space(@class), ' '), ' table-wrap ')][1]");
  await wrap.focus();
  if (!(await wrap.evaluate((el) => document.activeElement === el))) return -1; // the region did not take focus
  await page.keyboard.press(lang === "ar" ? "ArrowLeft" : "ArrowRight");
  await page.keyboard.press(lang === "ar" ? "ArrowLeft" : "ArrowRight");
  // Chromium animates keyboard scrolling: poll up to 5 s for the change (as the round-3 probe, revision 3).
  let moved = 0;
  for (let i = 0; i < 25 && moved < 1; i++) {
    await page.waitForTimeout(200);
    moved = Math.abs(await wrap.evaluate((el) => el.scrollLeft));
  }
  return moved;
}

async function sweep(
  page: Page,
  lang: Lang,
  name: string,
  scope: string,
  tableSel: string,
  open: () => Promise<void>,
  failures: string[],
  out: Record<string, unknown>,
) {
  for (const [label, width, font] of SIZES) {
    await page.setViewportSize({ width, height: 900 });
    await open();
    if (font) await page.evaluate((f) => (document.documentElement.style.fontSize = `${f}%`), font);
    await page.waitForLoadState("networkidle");
    await page.waitForTimeout(300);
    const where = `${name} ${label}`;
    const m = await measure(page, scope);
    const target = page.locator(tableSel).first();
    if ((await target.count()) === 0) failures.push(`${where}: set-up: ${tableSel} not rendered`);
    if (m.pageOverflow > 1) failures.push(`${where}: page scrolls ${m.pageOverflow} px`);
    if (m.escapes.length) failures.push(`${where}: elements outside the viewport, not in a scroll container: ${m.escapes.join(" | ")}`);
    if (m.dialog && (m.dialog.left < -1 || m.dialog.right > m.W + 1)) failures.push(`${where}: dialog [${m.dialog.left}..${m.dialog.right}] of ${m.W}`);
    for (const t of m.tables) {
      if (!t.region) failures.push(`${where}: table ${t.sel} is in no .table-wrap region`);
      else {
        if (t.region.tabIndex !== 0 || t.region.role !== "region" || !t.region.name) failures.push(`${where}: table ${t.sel} region not focusable/named ${JSON.stringify(t.region)}`);
        if (t.region.left < -1 || t.region.right > m.W + 1) failures.push(`${where}: table ${t.sel} region [${t.region.left}..${t.region.right}] of ${m.W}`);
      }
    }
    let kb: number | null = null;
    const tgt = m.tables.find((t) => t.region?.scrolls);
    if (width !== 1280 && tgt && (await target.count()) > 0) {
      kb = await keyboardScroll(page, lang, target);
      if (kb === -1) failures.push(`${where}: the region does not take keyboard focus`);
      else if (kb < 1) failures.push(`${where}: the focused region does not scroll with the arrow key`);
    }
    if (label === "390") {
      const axe = await new AxeBuilder({ page }).withTags(["wcag2a", "wcag2aa", "wcag21a", "wcag21aa"]).analyze();
      const serious = axe.violations.filter((v) => v.impact === "serious" || v.impact === "critical");
      if (serious.length) failures.push(`${where}: axe ${serious.map((v) => `${v.id}(${v.nodes.length})`).join(", ")}`);
      out[`${name}/axe`] = axe.violations.map((v) => `${v.id}:${v.impact}:${v.nodes.length}`);
    }
    out[`${name}/${label}`] = { ...m, keyboardScrolled: kb };
    await shot(page, lang, `r5-${name.replace(/[^a-z0-9]+/gi, "-").toLowerCase()}-${label}`);
    if (font) await page.evaluate(() => (document.documentElement.style.fontSize = ""));
  }
}

function save(lang: Lang, file: string, out: Record<string, unknown>) {
  mkdirSync(path.join(EVIDENCE, lang), { recursive: true });
  writeFileSync(path.join(EVIDENCE, lang, file), JSON.stringify(out, null, 1));
}

test("R5-FB: Form builder field table (long extension field draft; auditor read-only) at 390, 320 px and 200 % text", async ({ browser, page, playwright }, info) => {
  test.setTimeout(300_000);
  const lang = langOf(info);
  const admin = await apiSession(playwright, "dev.admin");
  type Assignment = { id: string; roleCode: string; revokedAt: string | null; version: number };
  const active = async () =>
    (await admin.call<{ items: Assignment[] }>("GET", `/api/v1/role-assignments?organizationId=${ORG}&userId=${NOBODY}`)).items.filter(
      (a) => a.roleCode === "ADM_METHOD" && a.revokedAt === null,
    );
  if ((await active()).length === 0)
    await admin.call("POST", "/api/v1/role-assignments", {
      userId: NOBODY,
      roleCode: "ADM_METHOD",
      scope: { type: "organization", id: ORG },
      reason: "Synthetic methodology administrator for the qa round-5 reflow probe",
    });
  const adm = await apiSession(playwright, "dev.nobody");
  type Version = { id: string; status: string; draftStage: string | null; isNewestPublished: boolean; version: number };
  const failures: string[] = [];
  const out: Record<string, unknown> = {};
  let draftId: string | null = null;
  try {
    for (const d of (await adm.call<{ items: Version[] }>("GET", `${V}?kind=form_schema&status=draft`)).items)
      if (d.draftStage === "editing" || d.draftStage === "preview") await adm.call("POST", `${V}/${d.id}/discard`, undefined, { ifMatch: d.version });
    const base = (await adm.call<{ items: Version[] }>("GET", `${V}?kind=form_schema&status=published`)).items.find((v) => v.isNewestPublished)!;
    draftId = (await adm.call<Version>("POST", V, { basedOnVersionId: base.id, changeSummary: `Synthetic qa r5 reflow (${lang})` })).id;
    await signIn(page, lang, "dev.nobody");
    const openDraft = async () => {
      await page.goto("/admin/forms/T03");
      await page.getByLabel(fieldLabel(lang, "forms.schema.version")).selectOption(draftId!);
      await expect(page.locator("[data-state='editable-draft']")).toBeVisible();
    };
    await openDraft();
    const key = `qa_r5_${lang}`;
    await page.getByRole("button", { name: tr(lang, "forms.schema.addField"), exact: true }).click();
    const add = page.getByRole("dialog");
    await add.getByLabel(fieldLabel(lang, "forms.schema.fieldKey")).fill(key);
    await add.getByLabel(fieldLabel(lang, "forms.schema.labelEn")).fill(`QA ${LONG_WORD}`);
    await add.getByLabel(fieldLabel(lang, "forms.schema.labelAr")).fill(LONG_AR);
    await add.getByRole("button", { name: tr(lang, "studio.elements.save"), exact: true }).click();
    await expect(add).toHaveCount(0);
    await expect(page.locator(`[data-field='${key}']`)).toBeVisible();
    await sweep(page, lang, "fb-draft", "main#main", "table[data-field-count]", openDraft, failures, out);
    const aud = await asUser(browser, lang, "dev.auditor");
    await sweep(
      aud.page,
      lang,
      "fb-readonly",
      "main#main",
      "table[data-field-count]",
      async () => {
        await aud.page.goto("/admin/forms/T03");
        await aud.page.getByLabel(fieldLabel(lang, "forms.schema.version")).selectOption(base.id);
        await expect(aud.page.locator("[data-state='read-only-version']")).toBeVisible();
        await expect(aud.page.locator("table[data-field-count] tbody tr").first()).toBeVisible();
      },
      failures,
      out,
    );
    await aud.close();
  } finally {
    if (draftId) {
      const d = await adm.call<Version>("GET", `${V}/${draftId}`);
      if (d.status === "draft") await adm.call("POST", `${V}/${draftId}/discard`, undefined, { ifMatch: d.version });
    }
    for (const a of await active()) await admin.call("POST", `/api/v1/role-assignments/${a.id}/revoke`, { reason: "End of the qa r5 probe" }, { ifMatch: a.version });
  }
  save(lang, "r5-fb.json", out);
  console.info(`R5-FB ${lang}: ${JSON.stringify(out)}`);
  expect(failures).toEqual([]);
});

async function evidence(lead: ApiSession, T: string, title: string) {
  await lead.call("POST", `${T}/evidence`, { kind: "note", title, noteBody: "Synthetic extract (demo data).", ownerUserId: lead.userId });
}

async function attach(scope: Locator, lang: Lang, title: string) {
  await scope.getByRole("tab", { name: tr(lang, "forms.framework.tabs.attachments"), exact: true }).click();
  await scope.getByLabel(tr(lang, "forms.attachments.choose"), { exact: true }).selectOption({ label: title });
  await scope.getByRole("button", { name: tr(lang, "forms.attachments.attach"), exact: true }).click();
  await expect(scope.locator("[data-attachment-count='1'] [data-status='active']")).toContainText(title.slice(0, 30));
}
