// DOMAIN REVIEWER PROBE (DG6 round 10), disposable clone only (copied as e2e/zz-domain/dom-r10-editor-tokens.spec.ts).
// SYNTHETIC data. Production-initialized dev stack (apps/web/e2e/support/with-stack.sh, no worker), chromium-en and
// chromium-ar. D-231 through the REAL rule editor, by a synthetic ADM_METHOD author not admitted to sensitive records.
//   A. F-DG6-580 re-run: [neq P, neq S, neq Q] (S sensitive); "Remove condition 1", "Save the draft", "Save the draft"
//      again. Expected now: both saves 200, the form reloads after each, S is still tested.
//   B. A concurrent change behind the open editor, then "Save the draft": 409 -> the bilingual "changed, re-apply"
//      alert, the form reloads, nothing hidden is lost.
//   C. PUBLISHED -> DRAFT: v1 [neq P, neq S, neq H] (S, H sensitive) published through the real API state machine
//      (dry run, submit, independent review, publish). Draft v2 is created; in the editor the author removes the S
//      condition and saves three times (the draft's row version reaches v1's). The author then restores "exclude S"
//      by copying the token shown for S on the published version 1 page into a new condition of draft v2.
// Marking a record sensitive or reviewing a rule version is an access/configuration decision, never a business approval
// (G1-G6). Nothing here touches DG0-DG7.
import { expect, test, type Page } from "@playwright/test";
import { apiSession, langOf, shot, signIn, SYN_RETAIL, tr } from "../../apps/web/e2e/support/ui.ts";
import { syntheticUser } from "../../apps/web/e2e/support/p3-journey-setup.ts";
import { psql } from "../support/a13-a22-stack.ts";

const ORG = "01920000-0000-7000-9000-000000000001";
const log = (s: string) => console.log(`DOMAIN-PROBE R10 ${s}`);
test.describe.configure({ timeout: 900_000 });
// Round-10 run 1 hung 15 min on innerText() of an absent alert (a probe bug, not a product result): every action is
// now bounded and the alert is read only when present.
test.use({ actionTimeout: 60_000 });

test("R10: version-bound tokens in the real rule editor", async ({ page, playwright }, info) => {
  const lang = langOf(info);
  const stamp = Date.now().toString(36);
  const admin = await apiSession(playwright, "dev.admin");
  const lead = await apiSession(playwright, "dev.lead");
  const office = await apiSession(playwright, "dev.office");
  const tid = (await lead.call<{ id: string }>("POST", "/api/v1/transformations", {
    businessUnitId: SYN_RETAIL, name: `Synthetic domain R10 rule ${lang.toUpperCase()} ${stamp}`, mode: "end_to_end" })).id;
  const T = `/api/v1/transformations/${tid}`;
  const bo = await syntheticUser(admin, ORG, tid, `dev.domr10.bo.${lang}.${stamp}`, `Synthetic Benefit Owner R10 ${lang}`, "BO", lang);
  const boApi = await apiSession(playwright, bo.username);
  const names = new Map<string, string>();
  const mk = async (n: string) => {
    const b = await boApi.call<{ id: string }>("POST", `${T}/benefits`, {
      title: `Synthetic DOMR10 ${n} ${lang} ${stamp}`, description: "Synthetic (domain probe).", benefitType: "revenue",
      valueClass: "revenue_uplift", ownerUserId: bo.id, currency: "SAR", financialStatementLine: "Revenue - roaming" });
    names.set(b.id, n);
    return b.id;
  };
  const mask = (s: string) => { for (const [id, n] of names) s = s.replaceAll(id, `<${n}>`); return s; };
  const P = await mk("P"), S = await mk("S"), Q = await mk("Q"), H = await mk("H");
  const methodUser = async (suffix: string) => {
    const username = `dev.domr10.${suffix}.${lang}.${stamp}`;
    const id = (await syntheticUser(admin, ORG, tid, username, `Synthetic ${suffix} R10 ${lang}`, "BO", lang)).id;
    await admin.call("POST", "/api/v1/role-assignments", { userId: id, roleCode: "ADM_METHOD", scope: { type: "organization", id: ORG },
      reason: "Synthetic rule author/reviewer for the domain probe (approves nothing real)" });
    return { username, id, api: await apiSession(playwright, username) };
  };
  const author = await methodUser("method");
  const R = "/api/v1/admin/automation-rules";
  // Rule C is transformation-scoped so that its dry run can plan the create_task action (run 2: an organization-scoped
  // rule's dry run had no transformation for the TL role and returned outcome "fail", so submit was refused).
  const newRule = (conds: string[], scope: Record<string, string> = { scopeLevel: "organization" }) => author.api.call<{ id: string }>("POST", R, {
    code: `test.domr10.${lang}.${Math.random().toString(36).slice(2, 8)}`, nameEn: `Synthetic exclusion rule R10 ${stamp}`,
    nameAr: `قاعدة استثناء تجريبية ${stamp}`, ...scope, organizationId: ORG, ownerUserId: author.id,
    content: { triggerCodes: ["benefit.variance_evaluated"], conditions: conds.map((v) => ({ field: "benefitId", op: "neq", value: v })),
      actions: [{ type: "create_task", parameters: { assigneeRole: "TL", messageKey: "automation.message.follow_up" } }], runAs: "service" } });
  const ruleA = await newRule([P, S, Q]);
  const ruleC = await newRule([P, S, H], { scopeLevel: "transformation", transformationId: tid });
  for (const b of [S, H])
    await office.call("PUT", `${T}/records/benefit/${b}/sensitivity`, { isSensitive: true, reason: "Synthetic commercially sensitive benefit" }, { ifMatch: 0 });
  const stored = (ruleId: string, no: number) =>
    mask(psql(`SELECT version || ' ' || conditions::text FROM automation_rule_version WHERE automation_rule_id = '${ruleId}' AND version_no = ${no}`)[0]!);
  const label = tr(lang, "access.includesRestrictedRecords");
  const form = (p: Page) => p.locator('[data-form="edit-rule-content"]');
  const screen = async (step: string) => {
    await page.waitForTimeout(1200);
    const values = await page.locator('[data-form="edit-rule-content"] [data-condition-row] input').evaluateAll((els) => els.map((e) => (e as HTMLInputElement).value)).catch(() => []);
    const text = await page.locator("main#main").innerText().catch(() => "");
    const badges = await page.locator('[data-restricted-records="true"]').count();
      const alertEl = page.locator('[data-state="rule-changed-reapply"]');
    const alert = (await alertEl.count()) > 0 ? await alertEl.innerText() : "";
    log(`${lang} ${step} url=${page.url().replace(/.*\/admin/, "/admin")} formVersion=${(await form(page).count()) > 0 ? await form(page).getAttribute("data-form-version") : "(no form)"} EDITOR values=${mask(JSON.stringify(values))} badges=${badges} labelCount=${text.split(label).length - 1} secretTitle=${/DOMR10 [SH] /.test(text)} namesS=${text.includes(S)} alert=${JSON.stringify(alert)}`);
    await shot(page, lang, `dom-r10-${step}`);
  };
  page.on("response", async (r) => {
    if (r.request().method() === "PATCH" && r.url().includes("/automation-rules/"))
      log(`${lang} PATCH ${r.url().replace(/.*automation-rules\//, "")} If-Match=${r.request().headers()["if-match"] ?? ""} body=${mask(r.request().postData() ?? "")} -> ${r.status()}`);
  });
  await page.context().clearCookies();
  await signIn(page, lang, author.username);
  const save = async () => {
    await form(page).getByRole("button", { name: tr(lang, "workflowsP5.rules.saveDraft") }).click();
    await page.waitForLoadState("networkidle");
    await page.waitForTimeout(800);
  };

  // ---- A. F-DG6-580 re-run
  await page.goto(`/admin/automation-rules/${ruleA.id}/versions/1`);
  await page.waitForLoadState("networkidle");
  await expect(page.locator("html")).toHaveAttribute("dir", lang === "ar" ? "rtl" : "ltr");
  await expect(form(page)).toBeVisible({ timeout: 30_000 });
  await screen("A1-read");
  await page.getByRole("button", { name: tr(lang, "workflowsP5.rules.removeCondition", { n: 1 }) }).click();
  await save();
  await screen("A2-after-remove-and-save");
  log(`${lang} A stored after save 1: ${stored(ruleA.id, 1)}`);
  await save();
  await screen("A3-after-second-save");
  log(`${lang} A stored after save 2: ${stored(ruleA.id, 1)}`);
  expect.soft(stored(ruleA.id, 1), "A: S still tested after the second save").toContain("<S>");

  // ---- B. 409 behind the open editor -> bilingual re-apply alert
  await page.reload();
  await page.waitForLoadState("networkidle");
  const cur = await author.api.call<{ version: number; content: { conditions: unknown[] } } & Record<string, unknown>>("GET", `${R}/${ruleA.id}/versions/1`);
  await author.api.call("PATCH", `${R}/${ruleA.id}/versions/1`, { content: { ...cur.content, conditions: [...cur.content.conditions, { field: "severity", op: "eq", value: "high" }] } }, { ifMatch: cur.version });
  await page.getByRole("button", { name: tr(lang, "workflowsP5.rules.removeCondition", { n: 2 }) }).click();
  await save();
  await screen("B1-after-409");
  log(`${lang} B stored: ${stored(ruleA.id, 1)}`);
  await expect.soft(page.locator('[data-state="rule-changed-reapply"]'), "B: the re-apply alert").toHaveText(tr(lang, "workflowsP5.rules.changedReapply"));
  expect.soft(stored(ruleA.id, 1), "B: S still tested").toContain("<S>");

  // ---- C. published v1 -> draft v2
  const V1 = `${R}/${ruleC.id}/versions/1`;
  const sample = { triggerCode: "benefit.variance_evaluated", payload: { benefitId: Q, measurementId: crypto.randomUUID(), periodStart: "2026-01-01",
    periodEnd: "2026-03-31", plannedAmount: "1000000.00", measuredAmount: "850000.00", variance: "-150000.00", offTrack: true } };
  const dry = await author.api.call<{ outcome: string }>("POST", `${V1}/dry-runs`, sample);
  log(`${lang} C dry run outcome=${dry.outcome} ${mask(JSON.stringify(dry)).slice(0, 600)}`);
  let v = await author.api.call<{ version: number }>("GET", V1);
  await author.api.call("POST", `${V1}/submit`, undefined, { ifMatch: v.version });
  v = await author.api.call<{ version: number }>("GET", V1);
  // Run 3: a second ADM_METHOD user is 403 on review; the Transformation Office reviews, as in the candidate's a13 spec.
  await office.call("POST", `${V1}/review`, { outcome: "approve", note: "Synthetic configuration review (domain probe)." }, { ifMatch: v.version });
  const rule = await author.api.call<{ version: number }>("GET", `${R}/${ruleC.id}`);
  await author.api.call("POST", `${V1}/publish`, undefined, { ifMatch: rule.version });
  const rule2 = await author.api.call<{ version: number }>("GET", `${R}/${ruleC.id}`);
  await author.api.call("POST", `${R}/${ruleC.id}/versions`, { changeSummary: "Synthetic draft v2 (domain probe R10)" }, { ifMatch: rule2.version });
  log(`${lang} C v1 stored: ${stored(ruleC.id, 1)} | v2 stored: ${stored(ruleC.id, 2)}`);
  await page.goto(`/admin/automation-rules/${ruleC.id}/versions/1`);
  await page.waitForLoadState("networkidle");
  await screen("C1-published-v1");
  const v1Text = await page.locator('[data-content="read-only"] [data-condition="1"]').innerText();
  const tokenOfS = /\[restricted:\d+:\d+\]/.exec(v1Text)?.[0] ?? "";
  log(`${lang} C token shown for S on published v1: ${tokenOfS} (line: ${JSON.stringify(v1Text)})`);
  await page.goto(`/admin/automation-rules/${ruleC.id}/versions/2`);
  await page.waitForLoadState("networkidle");
  await expect(form(page)).toBeVisible({ timeout: 30_000 });
  await screen("C2-draft-v2-read");
  await page.getByRole("button", { name: tr(lang, "workflowsP5.rules.removeCondition", { n: 2 }) }).click();
  await save();
  await save();
  await save();
  await screen("C3-draft-v2-after-removing-S-and-3-saves");
  log(`${lang} C v2 stored before restore: ${stored(ruleC.id, 2)}`);
  await form(page).getByRole("button", { name: tr(lang, "workflowsP5.rules.addCondition") }).click();
  const row = form(page).locator("[data-condition-row]").last();
  await row.locator("select").nth(0).selectOption("benefitId");
  await row.locator("select").nth(1).selectOption("neq");
  await row.locator("input").fill(tokenOfS);
  await screen("C4-draft-v2-token-pasted");
  await save();
  await screen("C5-draft-v2-after-save");
  const after = stored(ruleC.id, 2);
  log(`${lang} C v2 stored after restoring S by its v1 token: ${after}`);
  await page.reload();
  await page.waitForLoadState("networkidle");
  await screen("C6-draft-v2-after-reload");
  expect.soft(after.includes("<S>") || !after.includes('"<H>"}, {"op": "neq", "field": "benefitId", "value": "<H>"'), "C: v1's token for S is refused or stands for S, never for H").toBe(true);
  expect.soft(after, "C: S restored (or the save refused)").toContain("<S>");
});
