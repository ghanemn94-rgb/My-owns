// DOMAIN REVIEWER PROBE (DG6 round 9), disposable clone only (copied as e2e/zz-domain/dom-r9-editor-stale-tokens.spec.ts).
// SYNTHETIC data. Production-initialized dev stack (apps/web/e2e/support/with-stack.sh, no worker), chromium-en and
// chromium-ar. D-226 / D-228 (1) through the REAL rule editor: a synthetic ADM_METHOD author (not admitted to sensitive
// records) owns a draft rule whose conditions are [benefitId neq P, benefitId neq S, benefitId neq Q]; the TO marks S
// sensitive. In the editor the author (1) reads the version, (2) clicks "Remove condition 1" and "Save draft",
// (3) clicks "Save draft" again without touching anything else. After each step: what the screen shows vs what is stored.
// Marking a record sensitive is an access decision, never a business approval (G1-G6). Nothing touches DG0-DG7.
import { expect, test } from "@playwright/test";
import { apiSession, langOf, shot, signIn, SYN_RETAIL, tr } from "../../apps/web/e2e/support/ui.ts";
import { syntheticUser } from "../../apps/web/e2e/support/p3-journey-setup.ts";
import { psql } from "../support/a13-a22-stack.ts";

const ORG = "01920000-0000-7000-9000-000000000001";
const log = (s: string) => console.log(`DOMAIN-PROBE R9 ${s}`);
test.describe.configure({ timeout: 600_000 });

test("R9: the rule editor saves the same form twice after a token position shift", async ({ page, playwright }, info) => {
  const lang = langOf(info);
  const stamp = Date.now().toString(36);
  const admin = await apiSession(playwright, "dev.admin");
  const lead = await apiSession(playwright, "dev.lead");
  const office = await apiSession(playwright, "dev.office");
  const tid = (await lead.call<{ id: string }>("POST", "/api/v1/transformations", {
    businessUnitId: SYN_RETAIL, name: `Synthetic domain R9 rule ${lang.toUpperCase()} ${stamp}`, mode: "end_to_end" })).id;
  const T = `/api/v1/transformations/${tid}`;
  const bo = await syntheticUser(admin, ORG, tid, `dev.domr9.bo.${lang}.${stamp}`, `Synthetic Benefit Owner R9 ${lang}`, "BO", lang);
  const boApi = await apiSession(playwright, bo.username);
  const mk = (title: string) => boApi.call<{ id: string }>("POST", `${T}/benefits`, {
    title, description: "Synthetic (domain probe).", benefitType: "revenue", valueClass: "revenue_uplift",
    ownerUserId: bo.id, currency: "SAR", financialStatementLine: "Revenue - roaming" });
  const S = await mk(`Synthetic DOMR9 SECRET ${lang} ${stamp}`);
  const P = await mk(`Synthetic DOMR9 open P ${lang} ${stamp}`);
  const Q = await mk(`Synthetic DOMR9 open Q ${lang} ${stamp}`);
  const authorName = `dev.domr9.method.${lang}.${stamp}`;
  const user = await admin.call<{ id: string }>("POST", "/api/v1/users", { organizationId: ORG, displayName: `Synthetic rule author R9 ${lang}`,
    preferredLocale: lang, identity: { issuer: "urn:mth:dev-local", subject: authorName } }).catch(async () => null);
  let authorId = user?.id;
  if (!authorId) authorId = (await syntheticUser(admin, ORG, tid, authorName, `Synthetic rule author R9 ${lang}`, "BO", lang)).id;
  await admin.call("POST", "/api/v1/role-assignments", { userId: authorId, roleCode: "ADM_METHOD", scope: { type: "organization", id: ORG },
    reason: "Synthetic rule author for the domain probe (approves nothing real)" });
  const author = await apiSession(playwright, authorName);
  const rule = await author.call<{ id: string }>("POST", "/api/v1/admin/automation-rules", {
    code: `test.domr9.${lang}.${stamp}`, nameEn: `Synthetic exclusion rule R9 ${stamp}`, nameAr: `قاعدة استثناء تجريبية ${stamp}`,
    scopeLevel: "organization", organizationId: ORG, ownerUserId: authorId,
    content: { triggerCodes: ["benefit.variance_evaluated"],
      conditions: [{ field: "benefitId", op: "neq", value: P.id }, { field: "benefitId", op: "neq", value: S.id }, { field: "benefitId", op: "neq", value: Q.id }],
      actions: [{ type: "create_task", parameters: { assigneeRole: "TL", messageKey: "automation.message.follow_up" } }], runAs: "service" } });
  await office.call("PUT", `${T}/records/benefit/${S.id}/sensitivity`, { isSensitive: true, reason: "Synthetic commercially sensitive benefit" }, { ifMatch: 0 });
  const stored = () => psql(`SELECT conditions::text FROM automation_rule_version WHERE automation_rule_id = '${rule.id}' AND version_no = 1`)[0]!
    .replaceAll(S.id, "<S>").replaceAll(P.id, "<P>").replaceAll(Q.id, "<Q>");
  const label = tr(lang, "access.includesRestrictedRecords");
  const screen = async (step: string) => {
    await page.waitForTimeout(1500);
    const values = await page.locator('[data-form="edit-rule-content"] [data-condition-row] input').evaluateAll((els) => els.map((e) => (e as HTMLInputElement).value));
    const text = await page.locator("main#main").innerText().catch(() => "");
    const shown = values.map((v) => v.replaceAll(P.id, "<P>").replaceAll(Q.id, "<Q>").replaceAll(S.id, "<S>"));
    const badges = await page.locator('[data-form="edit-rule-content"] [data-restricted-records="true"]').count();
    log(`${lang} ${step} EDITOR values=${JSON.stringify(shown)} badges=${badges} labelCount=${text.split(label).length - 1} secretTitle=${text.includes("SECRET")} hasS=${text.includes(S.id)}`);
    log(`${lang} ${step} DB stored: ${stored()}`);
    await shot(page, lang, `dom-r9-editor-${step}`);
  };
  const patches: string[] = [];
  page.on("request", (r) => {
    if (r.method() === "PATCH" && r.url().includes(`/automation-rules/${rule.id}/versions/1`))
      patches.push(`${r.headers()["if-match"] ?? ""} ${(r.postData() ?? "").replaceAll(P.id, "<P>").replaceAll(Q.id, "<Q>").replaceAll(S.id, "<S>")}`);
  });
  page.on("response", (r) => {
    if (r.request().method() === "PATCH" && r.url().includes(`/automation-rules/${rule.id}/versions/1`)) log(`${lang} PATCH -> ${r.status()}`);
  });
  await page.context().clearCookies();
  await signIn(page, lang, authorName);
  await page.goto(`/admin/automation-rules/${rule.id}/versions/1`);
  await page.waitForLoadState("networkidle");
  await expect(page.locator("html")).toHaveAttribute("dir", lang === "ar" ? "rtl" : "ltr");
  await expect(page.locator('[data-form="edit-rule-content"]')).toBeVisible({ timeout: 30_000 });
  await screen("1-read");
  await page.getByRole("button", { name: tr(lang, "workflowsP5.rules.removeCondition", { n: 1 }) }).click();
  await page.locator('[data-form="edit-rule-content"]').getByRole("button", { name: tr(lang, "workflowsP5.rules.saveDraft") }).click();
  await page.waitForLoadState("networkidle");
  await screen("2-after-remove-and-save");
  await page.locator('[data-form="edit-rule-content"]').getByRole("button", { name: tr(lang, "workflowsP5.rules.saveDraft") }).click();
  await page.waitForLoadState("networkidle");
  await screen("3-after-second-save");
  for (const p of patches) log(`${lang} PATCH sent: ${p}`);
  await page.reload();
  await page.waitForLoadState("networkidle");
  await screen("4-after-reload");
  expect.soft(stored(), "the hidden S is still tested after the second save (or the second save was refused)").toContain("<S>");
});
